/* Funciona dentro de Apps Script (google.script.run) o como página normal (fetch al /exec) */
var API_URL = 'https://script.google.com/macros/s/AKfycbzOiwz0lWWgExbIeJ7nF7bwWm2GSPg7NRR3C3zZRh81RhQ6NHVqX9u8sN8eJKOlxd1N/exec';
var EN_APPS_SCRIPT = !!(window.google && google.script && google.script.run);
var CLAVE_TOKEN = 'glorin-portal-token';

var TOKEN = leerLocal(CLAVE_TOKEN);
var D = null;                 // datos que manda el servidor
var PARAMS = {};
var S = { tab: 'fechas', mes: '', dia: '', abiertos: {}, foco: '', verPasados: false, filtroPagos: 'pend', editando: null, sel: {} };

function leerLocal(k){ try { return localStorage.getItem(k) || ''; } catch (e) { return ''; } }
function guardarLocal(k, v){ try { if (v) localStorage.setItem(k, v); else localStorage.removeItem(k); } catch (e) {} }

function api(fn, args){
  args = args || [];
  return new Promise(function(ok, ko){
    if (EN_APPS_SCRIPT){
      var run = google.script.run.withSuccessHandler(ok).withFailureHandler(function(err){ ko(err instanceof Error ? err : new Error(String(err && err.message || err))); });
      run[fn].apply(run, args);
      return;
    }
    fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ accion: 'portal', fn: fn, args: args }) })
      .then(function(r){ return r.json(); })
      .then(function(j){ if (j.ok) ok(j.data); else ko(new Error(j.error)); })
      .catch(function(){ ko(new Error('No hay conexión. Probá de nuevo.')); });
  });
}

function leerParams(cb){
  if (EN_APPS_SCRIPT && google.script.url) { google.script.url.getLocation(function(l){ cb(l.parameter || {}); }); return; }
  var p = {}; new URLSearchParams(location.search).forEach(function(v, k){ p[k] = v; }); cb(p);
}

/* ===== utilidades ===== */
function e(s){ return String(s == null ? '' : s).replace(/[&<>"']/g, function(c){ return { '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[c]; }); }
function pesos(n){ return '$' + Math.round(Number(n) || 0).toString().replace(/\B(?=(\d{3})+(?!\d))/g, '.'); }

function partes(iso){ return { y: +iso.slice(0,4), m: +iso.slice(5,7) - 1, d: +iso.slice(8,10) }; }
function iso(y, m, d){ return y + '-' + String(m + 1).padStart(2, '0') + '-' + String(d).padStart(2, '0'); }
function $(id){ return document.getElementById(id); }

function toast(msg, err){
  var t = $('toast'); t.textContent = msg; t.className = 'toast' + (err ? ' err' : ''); t.style.display = 'block';
  clearTimeout(toast._t); toast._t = setTimeout(function(){ t.style.display = 'none'; }, err ? 6000 : 3800);
}
function bloquear(on){ $('bloq').classList.toggle('on', !!on); }

function manejarError(err){
  bloquear(false);
  if (err && err.message === 'SESION'){ salirLocal('Tu sesión venció. Ingresá de nuevo.'); return; }
  toast(err && err.message || 'Algo salió mal. Probá de nuevo.', true);
}

// Llama al servidor, actualiza los datos y vuelve a dibujar
function accion(fn, args, despues){
  bloquear(true);
  return api(fn, args).then(function(datos){
    bloquear(false);
    if (datos && datos.eventos){ D = datos; if (datos.foco) { S.foco = datos.foco; S.abiertos[datos.foco] = true; } }
    if (despues) despues(datos);
    dibujar();
    if (datos && datos.mensaje) toast(datos.mensaje);
    return datos;
  }).catch(manejarError);
}

/* ===== arranque ===== */
leerParams(function(p){
  PARAMS = p;
  if (p.reset) { pantallaReset(p.reset); return; }
  var foco = p.fecha ? Promise.resolve(p.fecha) : (p.convocatoria ? api('portalResolverConvocatoria', [p.convocatoria]).catch(function(){ return ''; }) : Promise.resolve(''));
  foco.then(function(id){
    if (id) { S.foco = id; S.abiertos[id] = true; }
    if (TOKEN) cargar(); else pantallaIngreso();
  });
});

function cargar(){
  api('portalDatos', [TOKEN]).then(function(datos){
    D = datos;
    if (S.foco){
      var ev = D.eventos.filter(function(x){ return x.id === S.foco; })[0];
      if (ev) { S.mes = ev.fecha.slice(0, 7); S.tab = (!D.yo.admin && ev.mio === 'asignado') ? 'mios' : 'fechas'; }
    }
    if (!S.mes) S.mes = D.hoy.slice(0, 7);
    dibujar();
    if (S.foco) setTimeout(function(){ var el = $('ev-' + S.foco); if (el) el.scrollIntoView({ behavior: 'smooth', block: 'start' }); }, 120);
  }).catch(function(err){
    if (err && err.message === 'SESION') { TOKEN = ''; guardarLocal(CLAVE_TOKEN, ''); pantallaIngreso('Tu sesión venció. Ingresá de nuevo.'); }
    else { $('app').innerHTML = '<div class="card"><h2>Ups</h2><p>' + e(err.message) + '</p><button class="btn pri" onclick="location.reload()">Reintentar</button></div>'; }
  });
}

function entrar(res){
  if (res && res.pendiente) { pantallaPendiente(); return; }
  TOKEN = res.token; guardarLocal(CLAVE_TOKEN, TOKEN);
  $('app').innerHTML = '<div class="cargando"><div class="spin"></div>Entrando…</div>';
  cargar();
}

function salirLocal(msg){
  if (TOKEN) api('portalSalir', [TOKEN]).catch(function(){});
  TOKEN = ''; guardarLocal(CLAVE_TOKEN, ''); D = null;
  pantallaIngreso(msg);
}

function tabsDisponibles(){
  if (D.yo.admin) return [['fechas','📅','Fechas'], ['nueva','➕','Nueva'], ['profes','👥','Profes'], ['pagos','💵','Pagos'], ['cuenta','🙂','Cuenta']];
  return [['fechas','📅','Fechas'], ['mios','⭐','Mis cumples'], ['pagos','💵','Pagos'], ['cuenta','🙂','Cuenta']];

function dibujar(){
  if (!D) return;
  $('yo').hidden = false; $('yo').textContent = (D.yo.admin ? '⚙️ ' : '') + D.yo.nombre;
  $('tabs').hidden = false;
  var pendientesProfes = D.yo.admin ? (D.profes || []).filter(function(p){ return /pend/i.test(p.estado); }).length : 0;
  var misProximos = D.eventos.filter(function(x){ return x.mio === 'asignado' && !x.pasado && !x.cancelado; }).length;
  $('tabs-in').innerHTML = tabsDisponibles().map(function(t){
    var badge = t[0] === 'profes' && pendientesProfes ? pendientesProfes : (t[0] === 'mios' && misProximos ? misProximos : 0);
    return '<button class="' + (S.tab === t[0] ? 'on' : '') + '" data-tab="' + t[0] + '"><span class="ico">' + t[1] + '</span>' + t[2] +
      (badge ? '<span class="badge">' + badge + '</span>' : '') + '</button>';
  }).join('');
  var titulos = { fechas: D.yo.admin ? 'Fechas y equipos' : 'Fechas', mios: 'Mis cumples', nueva: S.editando ? 'Editar fecha' : 'Nueva fecha', profes: 'Profes', pagos: 'Pagos', cuenta: 'Mi cuenta' };
  $('titulo').textContent = titulos[S.tab] || 'Portal de profes';
  var vistas = { fechas: vistaFechas, mios: vistaMios, nueva: vistaNueva, profes: vistaProfes, pagos: vistaPagos, cuenta: vistaCuenta };
  $('app').innerHTML = (vistas[S.tab] || vistaFechas)();
  if (S.tab === 'nueva') prepararFormulario();
}

  var t = ev.target.closest('[data-tab],[data-acc]');
  if (!t) return;
  if (t.dataset.tab){ S.tab = t.dataset.tab; if (S.tab !== 'nueva') S.editando = null; S.dia = ''; dibujar(); window.scrollTo(0, 0); return; }
  var a = t.dataset.acc, id = t.dataset.id;
  var acciones = {
    mes: function(){ var p = partes(S.mes + '-01'); var d = new Date(p.y, p.m + Number(t.dataset.delta), 1); S.mes = iso(d.getFullYear(), d.getMonth(), 1).slice(0, 7); S.dia = ''; dibujar(); },
    dia: function(){ S.dia = S.dia === t.dataset.dia ? '' : t.dataset.dia; dibujar(); },
    todas: function(){ S.dia = ''; dibujar(); },
    abrir: function(){ S.abiertos[id] = !S.abiertos[id]; dibujar(); },
    pasados: function(){ S.verPasados = !S.verPasados; dibujar(); },
    proponer: function(){ accion('portalPostular', [TOKEN, id, true]); },
    retirar: function(){ accion('portalPostular', [TOKEN, id, false]); },
    bajarme: function(){ if (confirm('¿Seguro que no podés ir? Le avisamos a GLORIN para buscar reemplazo.')) accion('portalPostular', [TOKEN, id, false]); },
    publicar: function(){ accion('portalPublicar', [TOKEN, id]); },
    equipo: function(){ guardarEquipo(id); },
    editar: function(){ S.editando = id; S.tab = 'nueva'; dibujar(); window.scrollTo(0, 0); },
    cancelarEdicion: function(){ S.editando = null; S.tab = 'fechas'; dibujar(); },
    aprobar: function(){ accion('portalEstadoProfe', [TOKEN, t.dataset.email, 'Sí']); },
    desactivar: function(){ if (confirm('¿Desactivar a este profe? No va a poder entrar ni le llegan convocatorias.')) accion('portalEstadoProfe', [TOKEN, t.dataset.email, 'No']); },
    pagado: function(){ accion('portalMarcarPago', [TOKEN, id, true]); },
    pendiente: function(){ accion('portalMarcarPago', [TOKEN, id, false]); },
    filtroPagos: function(){ S.filtroPagos = t.dataset.f; dibujar(); },

/* ===== Fechas (calendario + lista) ===== */
function claseDot(x){
  if (x.cancelado) return 'd-gris';
  if (D.yo.admin){ if (!x.publicada) return 'd-gris'; return x.asignados.length >= x.necesarios ? 'd-mia' : 'd-abierta'; }
  if (x.mio === 'asignado') return 'd-mia';
  if (x.mio) return 'd-prop';
  return 'd-abierta';
}

function eventosVisibles(){
  return D.eventos.filter(function(x){
    if (S.verPasados) return true;
    return !x.pasado && !(x.cancelado && !x.mio && !D.yo.admin);
  });
}

function calendario(lista){
  var p = partes(S.mes + '-01');
  var primero = new Date(p.y, p.m, 1), dias = new Date(p.y, p.m + 1, 0).getDate();
  var offset = (primero.getDay() + 6) % 7;   // semana arranca el lunes
  var porDia = {};
  lista.forEach(function(x){ (porDia[x.fecha] = porDia[x.fecha] || []).push(x); });
  var h = '<div class="card"><div class="cal-top"><button class="navbtn" data-acc="mes" data-delta="-1" aria-label="Mes anterior">‹</button>' +
    '<div class="mes">' + MESES[p.m] + ' ' + p.y + '</div><button class="navbtn" data-acc="mes" data-delta="1" aria-label="Mes siguiente">›</button></div><div class="cal">';
  ['L','M','M','J','V','S','D'].forEach(function(d){ h += '<div class="dow">' + d + '</div>'; });
  for (var i = 0; i < offset; i++) h += '<div class="dia fuera"></div>';
  for (var d = 1; d <= dias; d++){
    var k = iso(p.y, p.m, d), evs = porDia[k] || [];
    var cls = 'dia' + (k < D.hoy ? ' pasado' : '') + (k === D.hoy ? ' hoy' : '') + (evs.length ? ' con' : '') + (S.dia === k ? ' sel' : '');
    h += '<button class="' + cls + '"' + (evs.length ? ' data-acc="dia" data-dia="' + k + '"' : ' disabled') + '>' + d +
      '<span class="dots">' + evs.slice(0, 3).map(function(x){ return '<span class="dot ' + claseDot(x) + '"></span>'; }).join('') + '</span></button>';
  }
  h += '</div><div class="leyenda">' + (D.yo.admin
    ? '<span><i class="dot d-abierta"></i>Faltan profes</span><span><i class="dot d-mia"></i>Equipo completo</span><span><i class="dot d-gris"></i>Sin publicar</span>'
    : '<span><i class="dot d-abierta"></i>Buscan profes</span><span><i class="dot d-prop"></i>Te propusiste</span><span><i class="dot d-mia"></i>Vas</span>') + '</div></div>';
  return h;
}

function vistaFechas(){
  var lista = eventosVisibles();
  var h = calendario(lista);
  var mostrar = S.dia ? lista.filter(function(x){ return x.fecha === S.dia; }) : lista.filter(function(x){ return S.verPasados || x.fecha.slice(0, 7) >= S.mes || x.id === S.foco; });
  if (!D.yo.admin && !S.dia) mostrar = mostrar.filter(function(x){ return x.mio !== 'asignado' || x.id === S.foco; });
  h += '<div class="seccion"><h3>' + (S.dia ? diaLargo(S.dia) : (D.yo.admin ? 'Próximas fechas' : 'Fechas abiertas')) + '</h3>' +
    (S.dia ? '<button class="link" data-acc="todas">Ver todas</button>' : '<button class="link" data-acc="pasados">' + (S.verPasados ? 'Ocultar pasadas' : 'Ver pasadas') + '</button>') + '</div>';
  if (!mostrar.length){
    h += '<div class="card vacio"><div class="big">🎈</div>' + (D.yo.admin ? 'No hay fechas para mostrar. Creá una desde <b>➕ Nueva</b>.' : 'Por ahora no hay fechas abiertas. Te avisamos por mail cuando haya una nueva.') + '</div>';
  }
  return h + mostrar.map(tarjeta).join('');
}

function vistaMios(){
  var mios = D.eventos.filter(function(x){ return x.mio === 'asignado'; });
  var proximos = mios.filter(function(x){ return !x.pasado; }), pasados = mios.filter(function(x){ return x.pasado; }).reverse();
  var propuestos = D.eventos.filter(function(x){ return (x.mio === 'postulado' || x.mio === 'suplente') && !x.pasado; });
  var h = '<div class="seccion"><h3>Próximos cumples</h3></div>';
  h += proximos.length ? proximos.map(tarjeta).join('') : '<div class="card vacio"><div class="big">⭐</div>Todavía no tenés cumples confirmados. Proponete desde <b>📅 Fechas</b>.</div>';
  if (propuestos.length) h += '<div class="seccion"><h3>Esperando confirmación</h3></div>' + propuestos.map(tarjeta).join('');
  if (pasados.length) h += '<div class="seccion"><h3>Ya hechos</h3><span class="muted small">' + pasados.length + '</span></div>' + pasados.map(tarjeta).join('');
  return h;
}

function diaLargo(k){
  var p = partes(k), dias = ['domingo','lunes','martes','miércoles','jueves','viernes','sábado'];
  return dias[new Date(p.y, p.m, p.d).getDay()] + ' ' + p.d + ' de ' + MESES[p.m];
}

function fila(a, b){ return b || b === 0 ? '<div class="r"><span>' + a + '</span><b>' + e(b) + '</b></div>' : ''; }

function tarjeta(x){
  var p = partes(x.fecha), abierto = !!S.abiertos[x.id];
  var faltan = Math.max(0, x.necesarios - x.asignados.length);
  var tags = [];
  if (x.cancelado) tags.push('<span class="tag">❌ Cancelado</span>');
  if (x.mio === 'asignado') tags.push('<span class="tag ok">✅ Vas</span>');
  if (x.mio === 'postulado') tags.push('<span class="tag ambar">⏳ Te propusiste</span>');
  if (x.mio === 'suplente') tags.push('<span class="tag ambar">Suplente</span>');
  if (!x.cancelado && !x.pasado) tags.push(faltan ? '<span class="tag rojo">Faltan ' + faltan + ' de ' + x.necesarios + '</span>' : '<span class="tag ok">Equipo completo</span>');
  if (D.yo.admin){
    if (!x.publicada && !x.cancelado) tags.push('<span class="tag">Sin publicar</span>');
    if (x.estado && x.estado !== 'Señado' && !x.cancelado) tags.push('<span class="tag">' + e(x.estado) + '</span>');
    if (x.postulados && x.postulados.length) tags.push('<span class="tag ambar">🙋 ' + x.postulados.length + ' propuesto' + (x.postulados.length > 1 ? 's' : '') + '</span>');
  }
  if (x.pago) tags.push('<span class="tag">💵 ' + pesos(x.pago) + '</span>');

  var h = '<div class="ev' + (x.cancelado ? ' cancelado' : '') + (S.foco === x.id ? ' foco' : '') + '" id="ev-' + e(x.id) + '">' +
    '<button class="ev-cab" data-acc="abrir" data-id="' + e(x.id) + '" aria-expanded="' + abierto + '">' +
    '<div class="fecha-chip"><span class="n">' + p.d + '</span><span class="m">' + MES_CORTO[p.m] + '</span></div>' +
    '<div style="flex:1;min-width:0"><div class="ev-tit">Cumple de ' + e(x.cumpleanero) + (x.edad ? ' (' + e(x.edad) + ')' : '') + '</div>' +
    '<div class="ev-sub">' + e(diaLargo(x.fecha)) + ' · ' + e(x.hora) + ' a ' + e(x.horaFin) + ' hs' + (x.zona ? ' · ' + e(x.zona) : '') + '</div>' +
    '<div class="tags">' + tags.join('') + '</div></div>' +
    '<span class="muted" style="font-size:18px">' + (abierto ? '▴' : '▾') + '</span></button>';
  if (abierto) h += '<div class="ev-det">' + detalle(x) + '</div>';
  return h + '</div>';
}

function detalle(x){
  var h = fila('Llegada', x.llegada + ' hs (para armar)') +
    (x.direccion ? fila('Dirección', x.direccion) : fila('Zona', x.zona)) +
    fila('Chicos', (x.chicos || '?') + (x.edades ? ' · ' + x.edades : '')) +
    fila('Paquete', x.paquete) + fila('Extras', x.extras) +
    fila('Pago por profe', x.pago ? pesos(x.pago) : 'a definir') +
    fila('Equipo', x.asignados.length ? x.asignados.join(', ') : 'todavía sin armar') +
    (x.familia ? fila('Familia', x.familia) : '') +
    (D.yo.admin && x.notas ? fila('Notas', x.notas) : '');

  if (x.direccion && !x.cancelado){
    h += '<a class="btn chico gris" target="_blank" href="https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(x.direccion + (x.zona ? ', ' + x.zona : '')) + '">📍 Cómo llegar</a>';
    if (x.waFamilia && D.yo.admin) h += '<a class="btn chico wa" target="_blank" href="' + e(x.waFamilia) + '">💬 Familia</a>';
  }

  if (D.yo.admin) return h + panelAdmin(x);
  if (x.cancelado || x.pasado) return h;
  if (x.mio === 'asignado') return h + '<button class="btn gris" data-acc="bajarme" data-id="' + e(x.id) + '">No puedo ir</button>';
  if (x.mio) return h + '<p class="small muted" style="margin:12px 0 0">' + (x.mio === 'suplente' ? 'El equipo ya está armado: si alguien se baja, te avisamos.' : 'Cuando armemos el equipo te llega un mail.') + '</p>' +
    '<button class="btn gris" data-acc="retirar" data-id="' + e(x.id) + '">Sacar mi propuesta</button>';
  return h + '<button class="btn ok" data-acc="proponer" data-id="' + e(x.id) + '">🙋 Me propongo</button>' +
    (x.asignados.length >= x.necesarios ? '<p class="small muted" style="margin:8px 0 0;text-align:center">El equipo está completo, quedarías como suplente.</p>' : '');
}

function panelAdmin(x){
  var id = e(x.id);
  var h = '<div style="margin-top:4px"><button class="btn chico gris" data-acc="editar" data-id="' + id + '">✏️ Editar</button>';
  if (!x.publicada && !x.cancelado && !x.pasado) h += '<button class="btn chico pri" data-acc="publicar" data-id="' + id + '">📣 Publicar y avisar a los profes</button>';
  h += '</div>';
  if (x.cancelado) return h;
  if (!S.sel[x.id]) S.sel[x.id] = x.asignados.slice();
  var sel = S.sel[x.id];
  var activos = (D.profes || []).filter(function(p){ return p.activo; }).map(function(p){ return p.nombre; });
  var orden = [];
  (x.postulados || []).concat(x.asignados).forEach(function(n){ if (orden.indexOf(n) < 0) orden.push(n); });
  var otros = activos.filter(function(n){ return orden.indexOf(n) < 0; });
  var item = function(n, nota){
    var on = sel.indexOf(n) >= 0;
    return '<label class="pick"><input type="checkbox" data-pick="' + id + '" value="' + e(n) + '"' + (on ? ' checked' : '') + '>' + e(n) +
      '<small>' + nota + '</small></label>';
  };
  h += '<div class="equipo"><b>Elegí el equipo</b> <span class="muted small">(<span id="cnt-' + id + '">' + sel.length + '</span> de ' + x.necesarios + ')</span>';
  h += orden.length ? orden.map(function(n){ return item(n, x.asignados.indexOf(n) >= 0 ? 'en el equipo' : '🙋 se propuso'); }).join('')
                    : '<p class="small muted" style="margin:8px 0">Todavía no se propuso nadie.' + (x.publicada ? '' : ' Publicá la fecha para que les llegue.') + '</p>';
  if (otros.length) h += '<details style="margin-top:8px"><summary class="small" style="font-weight:700;cursor:pointer">Elegir otro profe (' + otros.length + ')</summary>' +
    otros.map(function(n){ return item(n, ''); }).join('') + '</details>';
  h += '<label for="pago-' + id + '">Pago por profe</label><input id="pago-' + id + '" type="number" inputmode="numeric" min="0" step="500" placeholder="A definir" value="' + (x.pago == null ? '' : x.pago) + '">' +
    '<button class="btn ok" data-acc="equipo" data-id="' + id + '">Guardar equipo y avisar por mail</button>' +
    '<p class="small muted" style="margin:8px 0 0">A los que elegís les llega la confirmación con dirección y contacto, y quedan invitados al evento. Los que se propusieron y no quedan, como suplentes.</p></div>';
  return h;
}

function guardarEquipo(id){
  var x = D.eventos.filter(function(v){ return v.id === id; })[0];
  var sel = S.sel[id] || [];
  if (sel.length > x.necesarios && !confirm('Elegiste ' + sel.length + ' profes y el cumple pide ' + x.necesarios + '. ¿Guardar igual?')) return;
  var pago = $('pago-' + id).value;
  accion('portalConfirmarEquipo', [TOKEN, id, sel, pago === '' ? '' : Number(pago)], function(){ delete S.sel[id]; });
}

function prepararFormulario(){
  var f = $('fn'); if (!f) return;
  var sumar = function(h, min){ var p = h.split(':'), t = (+p[0]) * 60 + (+p[1]) + min; t = Math.min(t, 23 * 60 + 59); return String(Math.floor(t / 60)).padStart(2, '0') + ':' + String(t % 60).padStart(2, '0'); };
  if (!$('n-fin').value && $('n-hora').value) $('n-fin').value = sumar($('n-hora').value, D.duracionMin || 150);
  $('n-hora').addEventListener('change', function(){ if (!S.editando) $('n-fin').value = sumar(this.value, D.duracionMin || 150); });
  $('n-paq').addEventListener('change', function(){
    var p = D.paquetes.filter(function(q){ return q.nombre === $('n-paq').value; })[0];
    if (!p) return;
    $('n-nec').value = p.profes;
    if (p.pagoProfe && !$('n-pago').value) $('n-pago').value = p.pagoProfe;
  });
  f.onsubmit = function(ev){
    ev.preventDefault();
    var g = function(id){ var el = $(id); return el ? el.value : ''; };
    var d = { id: S.editando || '', fecha: g('n-fecha'), hora: g('n-hora'), horaFin: g('n-fin'), cumpleanero: g('n-cumple'), edad: g('n-edad'),
              paquete: g('n-paq'), necesarios: g('n-nec'), pago: g('n-pago'), zona: g('n-zona'), direccion: g('n-dir'), adulto: g('n-adulto'),
              whatsapp: g('n-wa'), chicos: g('n-chicos'), edades: g('n-edades'), extras: g('n-extras'), notas: g('n-notas'), estado: g('n-estado'),
              publicar: !!($('n-pub') && $('n-pub').checked) };
    accion('portalGuardarFecha', [TOKEN, d], function(datos){
      S.editando = null; S.tab = 'fechas';
      if (datos && datos.foco){ var x = datos.eventos.filter(function(v){ return v.id === datos.foco; })[0]; if (x) { S.mes = x.fecha.slice(0, 7); S.dia = x.fecha; } }
    }).then(function(){ setTimeout(function(){ var el = S.foco && $('ev-' + S.foco); if (el) el.scrollIntoView({ behavior: 'smooth' }); }, 100); });
  };
}

/* ===== Profes (admin) ===== */
function vistaProfes(){
  var ps = D.profes || [];
  var pend = ps.filter(function(p){ return /pend/i.test(p.estado); });
  var act = ps.filter(function(p){ return p.activo; });
  var inact = ps.filter(function(p){ return !p.activo && !/pend/i.test(p.estado); });
  var wa = function(p){ var n = String(p.whatsapp || '').replace(/\D/g, ''); return n ? '<a class="mini ok" target="_blank" href="https://wa.me/' + (n.length === 10 ? '549' + n : n) + '" style="text-decoration:none">💬</a>' : ''; };
  var filaP = function(p, botones){
    return '<div class="fila"><div class="q"><b>' + e(p.nombre) + '</b><span>' + e(p.email || 'sin mail') + (p.whatsapp ? ' · ' + e(p.whatsapp) : '') + (p.cuenta ? '' : ' · sin cuenta') + '</span></div>' + wa(p) + botones + '</div>';
  };
  var h = '<div class="card"><p class="small muted" style="margin:0 0 8px">Pasales este link para que creen su cuenta:</p>' +
    '<input readonly value="' + e(API_URL + '?profes=1') + '" onclick="this.select()"></div>';
  if (pend.length) h += '<div class="card"><h2>Esperan aprobación (' + pend.length + ')</h2>' + pend.map(function(p){
    return filaP(p, (p.email ? '<button class="mini pri" data-acc="aprobar" data-email="' + e(p.email) + '">Aprobar</button>' : ''));
  }).join('') + '</div>';
  h += '<div class="card"><h2>Activos (' + act.length + ')</h2>' + (act.length ? act.map(function(p){
    return filaP(p, p.email && p.email !== D.yo.email ? '<button class="mini gris" data-acc="desactivar" data-email="' + e(p.email) + '">Desactivar</button>' : '');
  }).join('') : '<p class="muted">Todavía no hay profes activos.</p>') + '</div>';
  if (inact.length) h += '<div class="card"><h2>Inactivos (' + inact.length + ')</h2>' + inact.map(function(p){
    return filaP(p, p.email ? '<button class="mini ok" data-acc="aprobar" data-email="' + e(p.email) + '">Activar</button>' : '');
  }).join('') + '</div>';
  return h;
}

/* ===== Pagos ===== */
function vistaPagos(){
  var ps = D.pagos.slice().sort(function(a, b){ return b.fecha.localeCompare(a.fecha); });
  var pend = ps.filter(function(p){ return p.estado !== 'Pagado'; }), pag = ps.filter(function(p){ return p.estado === 'Pagado'; });
  var suma = function(l){ return l.reduce(function(a, p){ return a + (p.monto || 0); }, 0); };
  var h = '<div class="totales"><div class="tot"><div class="l">' + (D.yo.admin ? 'A pagar' : 'Te deben') + '</div><div class="v">' + pesos(suma(pend)) + '</div><div class="l">' + pend.length + ' cumple' + (pend.length === 1 ? '' : 's') + '</div></div>' +
    '<div class="tot ok"><div class="l">' + (D.yo.admin ? 'Pagado' : 'Cobraste') + '</div><div class="v">' + pesos(suma(pag)) + '</div><div class="l">' + pag.length + ' cumple' + (pag.length === 1 ? '' : 's') + '</div></div></div>';
  if (pend.some(function(p){ return !p.monto; })) h += '<div class="aviso info">Hay cumples con el pago “a definir”. ' + (D.yo.admin ? 'Ponele el monto desde la fecha (Pago por profe).' : '') + '</div>';
  h += '<div class="seg"><button class="' + (S.filtroPagos === 'pend' ? 'on' : '') + '" data-acc="filtroPagos" data-f="pend">Pendientes</button><button class="' + (S.filtroPagos === 'todos' ? 'on' : '') + '" data-acc="filtroPagos" data-f="todos">Todos</button></div>';
  var lista = S.filtroPagos === 'pend' ? pend : ps;
  if (!lista.length) return h + '<div class="card vacio"><div class="big">💵</div>' + (S.filtroPagos === 'pend' ? 'No hay pagos pendientes.' : 'Todavía no hay pagos registrados. Aparecen cuando quedás en el equipo de un cumple.') + '</div>';
  var item = function(p){
    var q = '<div class="q"><b>' + (D.yo.admin ? e(p.profe) : 'Cumple de ' + e(p.cumple)) + '</b><span>' + (D.yo.admin ? 'Cumple de ' + e(p.cumple) + ' · ' : '') + e(p.fecha ? diaLargo(p.fecha) : '') +
      (p.estado === 'Pagado' ? ' · ✅ pagado' + (p.fechaPago ? ' el ' + e(p.fechaPago.slice(8, 10) + '/' + p.fechaPago.slice(5, 7)) : '') : '') + '</span></div>' +
      '<div class="monto">' + (p.monto ? pesos(p.monto) : 'a definir') + '</div>';
    if (D.yo.admin) q += p.estado === 'Pagado' ? '<button class="mini gris" data-acc="pendiente" data-id="' + e(p.id) + '">Deshacer</button>' : '<button class="mini ok" data-acc="pagado" data-id="' + e(p.id) + '">Pagado</button>';
    return '<div class="fila">' + q + '</div>';
  };
  if (!D.yo.admin) return h + '<div class="card">' + lista.map(item).join('') + '</div>';
  // Admin: agrupado por profe
  var grupos = {};
  lista.forEach(function(p){ (grupos[p.profe] = grupos[p.profe] || []).push(p); });
  return h + Object.keys(grupos).sort().map(function(n){
    var g = grupos[n], debe = suma(g.filter(function(p){ return p.estado !== 'Pagado'; }));
    return '<div class="card"><div style="display:flex;justify-content:space-between;align-items:baseline"><h2 style="margin:0">' + e(n) + '</h2>' +
      (debe ? '<b style="color:var(--rojo)">' + pesos(debe) + '</b>' : '') + '</div>' + g.map(item).join('') + '</div>';
  }).join('');
}
