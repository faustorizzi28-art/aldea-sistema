  CFG.extras.forEach(function(e){ var c = D.extras[e.id]; if (c) filas.push([e.nombre + (c>1? ' ×' + c : ''), e.precio * c]); });
  if (z.recargo) filas.push(['Traslado ' + z.nombre, z.recargo]);
  var total = filas.reduce(function(a,f){ return a+f[1]; }, 0);
  return {filas:filas, total:total, paquete:p, zona:z};
}
function fechaLarga(k){
  if (!k) return '';
  var t = parseKey(k).toLocaleDateString('es-AR',{weekday:'long', day:'numeric', month:'long'});
  return t.charAt(0).toUpperCase() + t.slice(1);
}
function horaFin(h){ var m = aMin(h) + duracion(); return pad(Math.floor(m/60)) + ':' + pad(m%60); }

}

/* ===== Borrador (si cierran la página, al volver sigue todo cargado) ===== */
var CAMPOS = ['adulto','whatsapp','email','cumpleanero','edad','gustos','cantChicos','adultos','atencionEspecial','direccion','horaMerienda','horaTorta','musica','evitar','comentarios'];
var CLAVE_BORRADOR = 'glorin-reserva-v1';
var BORRADOR_LISTO = false;  // no guardar hasta haber leído el borrador anterior
function guardarBorrador(){
  if (!BORRADOR_LISTO) return;
  try {
    var o = { D: D, campos: {}, zona: $('zona').value, paso: paso, t: Date.now() };
    CAMPOS.forEach(function(i){ o.campos[i] = $(i).value; });
    localStorage.setItem(CLAVE_BORRADOR, JSON.stringify(o));
  } catch(e){}
}
function borrarBorrador(){ try { localStorage.removeItem(CLAVE_BORRADOR); } catch(e){} }
function cargarBorrador(){
  try {
    var o = JSON.parse(localStorage.getItem(CLAVE_BORRADOR) || 'null');
    if (!o || Date.now() - o.t > 30*86400000) { BORRADOR_LISTO = true; return; }
    Object.keys(o.D || {}).forEach(function(k){ D[k] = o.D[k]; });
    CAMPOS.forEach(function(i){ if (o.campos && o.campos[i] != null) $(i).value = o.campos[i]; });
    if (o.zona) $('zona').value = D.zona = o.zona;
    // Volver a marcar los chips
    document.querySelectorAll('.chips[data-campo]').forEach(function(g){
      var v = D[g.dataset.campo]; if (!v) return;
      var lista = Array.isArray(v) ? v : [v];
      g.querySelectorAll('.chip').forEach(function(c){ c.classList.toggle('sel', lista.indexOf(c.textContent) >= 0); });
    });
    // La fecha guardada puede haberse ocupado: si ya no está libre, se vuelve a elegir
    if (D.fecha && ['off','lleno'].indexOf(estadoDia(parseKey(D.fecha))) >= 0) { D.fecha = ''; D.hora = ''; D.extraMin = 0; }
    if (D.hora && !horaLibre(D.fecha, D.hora, duracion())) { D.hora = ''; D.extraMin = 0; }
    if (D.fecha){ var f = parseKey(D.fecha); mesVista = new Date(f.getFullYear(), f.getMonth(), 1); }
    $('borrador').style.display = 'flex';
    dibujarCal(); dibujarPaquetes(); actualizar();
    BORRADOR_LISTO = true;
    if (D.fecha && D.hora && o.paso > 1) irA(Math.min(o.paso, TOTAL_PASOS));
  } catch(e){}
  finally { BORRADOR_LISTO = true; }
}

function copiar(txt, btn){
  var ok = function(){ btn.textContent = '¡Copiado!'; setTimeout(function(){ btn.textContent = 'Copiar'; }, 1800); };
  if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(txt).then(ok, function(){ copiaVieja(txt); ok(); });
  else { copiaVieja(txt); ok(); }
}
